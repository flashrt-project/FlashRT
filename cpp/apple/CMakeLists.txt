cmake_minimum_required(VERSION 3.18)
if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Minimum macOS version")
endif()
project(flashrt_apple LANGUAGES CXX OBJCXX)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
  message(FATAL_ERROR "The Apple native pipeline requires an Apple Silicon host")
endif()
include(CTest)
include(GNUInstallDirs)
option(FLASHRT_TEST_APPLE_TENSOROPS "Run BF16 TensorOps GPU qualification" OFF)
message(STATUS "Apple native tests: BUILD_TESTING=${BUILD_TESTING}; FLASHRT_TEST_APPLE_TENSOROPS=${FLASHRT_TEST_APPLE_TENSOROPS}")
add_library(flashrt_apple STATIC src/pipeline.mm)
target_include_directories(flashrt_apple PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(flashrt_apple PUBLIC cxx_std_17)
set_target_properties(flashrt_apple PROPERTIES OBJCXX_STANDARD 17
  OBJCXX_STANDARD_REQUIRED YES POSITION_INDEPENDENT_CODE ON)
target_compile_options(flashrt_apple PRIVATE -fobjc-arc)
target_link_libraries(flashrt_apple PUBLIC "-framework Foundation" "-framework Metal")
add_library(flashrt_apple_plan SHARED src/plan.cpp)
target_link_libraries(flashrt_apple_plan PRIVATE flashrt_apple)
add_library(flashrt_apple_signpost SHARED src/signpost.mm)
set_target_properties(flashrt_apple_signpost PROPERTIES OBJCXX_STANDARD 17
  OBJCXX_STANDARD_REQUIRED YES)
foreach(target flashrt_apple flashrt_apple_plan flashrt_apple_signpost)
  set_target_properties(${target} PROPERTIES CXX_VISIBILITY_PRESET hidden
    OBJCXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endforeach()
set(FLASHRT_APPLE_EXPORT_CHECK "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/apple/check_exports.sh")
foreach(component plan signpost)
  set(target "flashrt_apple_${component}")
  set(exports "${CMAKE_CURRENT_SOURCE_DIR}/${component}.exports")
  target_link_options(${target} PRIVATE "LINKER:-exported_symbols_list,${exports}")
  set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS "${exports}" "${FLASHRT_APPLE_EXPORT_CHECK}")
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND bash "${FLASHRT_APPLE_EXPORT_CHECK}" "$<TARGET_FILE:${target}>" "${exports}"
    VERBATIM)
  if(BUILD_TESTING)
    add_test(NAME "apple_${component}_exports"
      COMMAND bash "${FLASHRT_APPLE_EXPORT_CHECK}" "$<TARGET_FILE:${target}>" "${exports}")
  endif()
endforeach()
add_executable(flashrt_apple_smoke tests/pipeline_smoke.cpp)
target_link_libraries(flashrt_apple_smoke PRIVATE flashrt_apple)
add_executable(flashrt_apple_bf16_smoke tests/bf16_gemm_smoke.cpp)
target_link_libraries(flashrt_apple_bf16_smoke PRIVATE flashrt_apple)
add_executable(flashrt_apple_attention_smoke tests/attention_smoke.cpp)
target_link_libraries(flashrt_apple_attention_smoke PRIVATE flashrt_apple)
add_executable(flashrt_apple_gemm_tune tests/gemm_tune.cpp)
target_link_libraries(flashrt_apple_gemm_tune PRIVATE flashrt_apple)
add_executable(flashrt_apple_kernel_registry_smoke tests/kernel_registry_smoke.cpp)
target_link_libraries(flashrt_apple_kernel_registry_smoke PRIVATE flashrt_apple)
# Kernel implementations belong to the hardware source tree; this directory
# owns the native runtime, C ABI adapter and validation executables.
set(FLASHRT_APPLE_KERNEL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../csrc/apple")
foreach(target flashrt_apple_plan flashrt_apple_bf16_smoke
               flashrt_apple_attention_smoke flashrt_apple_gemm_tune
               flashrt_apple_kernel_registry_smoke)
  target_include_directories(${target} PRIVATE "${FLASHRT_APPLE_KERNEL_ROOT}")
endforeach()
if(BUILD_TESTING)
  add_test(NAME apple_native_pipeline COMMAND flashrt_apple_smoke)
  add_test(NAME apple_native_pipeline_validation COMMAND flashrt_apple_smoke)
  set_tests_properties(apple_native_pipeline_validation PROPERTIES ENVIRONMENT "MTL_DEBUG_LAYER=1")
  # Explicit opt-in: the default runtime supports macOS 14; BF16 TensorOps
  # qualification needs a macOS 26.2+ host and must not silently skip there.
  if(FLASHRT_TEST_APPLE_TENSOROPS)
    add_test(NAME apple_kernel_registry COMMAND flashrt_apple_kernel_registry_smoke)
    set_tests_properties(apple_kernel_registry PROPERTIES ENVIRONMENT "MTL_DEBUG_LAYER=1")
    add_test(NAME apple_bf16_gemm COMMAND flashrt_apple_bf16_smoke)
    set_tests_properties(apple_bf16_gemm PROPERTIES ENVIRONMENT "MTL_DEBUG_LAYER=1")
    add_test(NAME apple_bf16_attention COMMAND flashrt_apple_attention_smoke)
    set_tests_properties(apple_bf16_attention PROPERTIES ENVIRONMENT "MTL_DEBUG_LAYER=1")
  endif()
endif()
install(TARGETS flashrt_apple ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS flashrt_apple_plan LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS flashrt_apple_signpost LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
